using Arena.Gameplay.View;
using UnityEngine;

namespace Arena.Editor.Art.Fighters.Recipes
{
    /// <summary>
    /// Tess, the garage inventor in a homemade exosuit (mobile assault) (A-Roster3D-unity2): FBX body, texture and weapon from <c>Art/Source/3D/tess/source.md</c>, installed like Theo through
    /// <see cref="FbxFighterRecipe"/>. The view of the data fighter <c>"tess"</c> (D-150).
    /// </summary>
    public static class TessRecipe
    {
        public const string FighterId = "tess";
        public const string Hero = "tess";
        /// <summary>The FBX is ≈ 1.0 tall (source.md); Theo stands 1.5.</summary>
        public const float Scale = 1.5f;

        public static FighterModelRecipe Create() => FbxFighterRecipe.Create(new FbxFighterSpec
        {
            FighterId = FighterId,
            Hero = Hero,
            Scale = Scale,
            AttackStyle = FighterAttackStyle.Shoot,
            UltStyle = FighterAttackStyle.Shoot,
            TrickStyle = FighterTrickStyle.Roll,
            LidRadii = new Vector3(0.030f, 0.022f, 0.010f),
            LidLift = -0.002f,
            LidSegments = 6
        });
    }
}
